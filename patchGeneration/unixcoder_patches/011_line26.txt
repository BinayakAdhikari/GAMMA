
public static <F, S> List<Pair<F, S>> hanoi(int height, int start, int end) {
    List<Pair<F, S>> steps = new ArrayList<Pair<F, S>>();
    for (int i = start; i <= end; i++) {
      Pair<F, S> step = new Pair<F, S>();
      step.first = i;
      step.second = i;
      steps.add(step);
    }
    return steps;
  }
